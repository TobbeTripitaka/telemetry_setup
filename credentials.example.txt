# TELE installer: example station settings and credentials
#
# This EXAMPLE belongs in Git. It contains no working passwords or tokens.
# Do not put real credentials in this tracked file, even temporarily.
# On the Ubuntu station, make a private copy outside the checkout:
#   sudo install -m 600 -o root -g root credentials.example.txt /root/tele-setup.conf
#   sudo nano /root/tele-setup.conf
#   sudo bash INSTALL.sh --config /root/tele-setup.conf --mode confirm
#
# Replace every REPLACE_WITH... value before installation. Password/key shapes
# shown in comments are deliberately fake examples, never recommended passwords.
# Literal KEY=value only: no wrapping quotes, shell expansion or inline comments.
# Keep JSON on ONE line. Comment out optional settings by starting the line with #.
# The installer splits this file into private runtime configuration files.
# For EMAIL ONLY after installation, see config/credentials.txt.example instead.

# STATION: a short unique label, not an email address or recorder serial.
# This selects <DROPBOX_ROOT>/tele/<station>/ in Dropbox.

OPERATOR_USER=tele

# DROPBOX: use the project account that has accepted the folder invitation.
# RCLONE_REMOTE is the NAME you assigned in "rclone config", without a colon.
# It is not a Dropbox email address, website URL or password.
RCLONE_REMOTE=tele_dropbox
DROPBOX_ROOT=/REPLACE_WITH_TEAM_FOLDER/REPLACE_WITH_PROJECT_FOLDER
# Example team path: /Research Team/Seismic Project
# Keep the leading / when addressing the team root; spaces are allowed.
# For a personal-space prefix, use the verified path without a leading /.
# A blank DROPBOX_ROOT uses the configured remote's default root.
#
# Choose ONE authorization method:
# A. Import an existing rclone config, root-owned and mode 0600 (recommended).
#    It must contain the RCLONE_REMOTE above. Import preserves namespace settings.
RCLONE_CONFIG_FILE=/root/REPLACE_WITH_YOUR_AUTHORIZED_RCLONE_CONFIG.conf
# B. Alternatively, comment out RCLONE_CONFIG_FILE and paste the COMPLETE JSON
#    returned by the rclone authorization flow. Do not enter a Dropbox password.
#    If custom client credentials/namespace settings are required, use method A.
#    The JSON below shows field names only; do not invent tokens or edit expiry.
# DROPBOX_TOKEN_JSON={"access_token":"REPLACE_WITH_RCLONE_ACCESS_TOKEN","token_type":"bearer","refresh_token":"REPLACE_WITH_RCLONE_REFRESH_TOKEN","expiry":"REPLACE_WITH_EXACT_EXPIRY_FROM_RCLONE"}
# On a rerun only, omit both to reuse the installed /etc/tele/rclone.conf.
# Authorization instructions: https://rclone.org/remote_setup/
# Shared/team folder details: https://rclone.org/dropbox/

# GMAIL: EMAIL_FROM is the account that SENDS the status email.
# EMAIL_TO is the recipient who should RECEIVE the report.
# These are unrelated to the Dropbox and Tailscale account identities.
EMAIL_FROM=REPLACE_WITH_SENDER@gmail.com
EMAIL_TO=REPLACE_WITH_REPORT_RECIPIENT@example.com
#
# EMAIL_PASSWORD is a GOOGLE APP PASSWORD, NOT your normal Google login password,
# NOT a one-time 2FA code, and NOT a Dropbox password.
# In the EMAIL_FROM Google account:
#   1. Enable 2-Step Verification.
#   2. Open App passwords and create one named "TELE station01" (use your station).
#   3. Copy its 16 characters here, removing any display spaces.
# Example shape ONLY: Google displays "abcd efgh ijkl mnop";
# the file would contain "abcdefghijklmnop" (without quotes).
# Never use that alphabetic demonstration as an actual password.
EMAIL_PASSWORD=REPLACE_WITH_YOUR_16_CHARACTER_GOOGLE_APP_PASSWORD
# Google revokes app passwords when you change the main Google account password.
# Generate a replacement and update the station if that happens.
# Some managed/Advanced Protection accounts may not offer app passwords.
# Instructions: https://support.google.com/accounts/answer/185833?hl=en

# PEGASUS: hardware identity, not credentials. Verify on the Ubuntu station:
#   lsblk -o NAME,TYPE,TRAN,SERIAL,MOUNTPOINTS
#   ls -l /dev/disk/by-id/
# Use the WHOLE recorder disk, never a -part1 path or the computer's OS disk.
RECORDER_DEVICE=/dev/disk/by-id/usb-REPLACE_WITH_REAL_RECORDER_ID
RECORDER_SERIAL=REPLACE_WITH_REAL_SERIAL
# Optional multi-recorder setup instead of the two lines above. New outer
# namespaces are pegasus_harvester/<id>/; this is a deliberate layout choice.
# SOURCES_JSON=[{"id":"pegasus01","device":"/dev/disk/by-id/usb-REPLACE_WITH_RECORDER_1_ID","serial":"REPLACE_WITH_SERIAL_1"},{"id":"pegasus02","device":"/dev/disk/by-id/usb-REPLACE_WITH_RECORDER_2_ID","serial":"REPLACE_WITH_SERIAL_2"}]

# NANOMETRICS PACKAGE: a vendor-authorized installer and its SHA256, not a password.
# Omit package fields only if the approved native binary is already installed.
HARVESTER_DEB=/root/REPLACE_WITH_APPROVED_PegasusHarvester.deb
HARVESTER_SHA256=REPLACE_WITH_TRUSTED_64_HEX_CHARACTER_PACKAGE_SHA256
# SHA256 shape: 64 hexadecimal characters (0-9 and a-f), with no filename/spaces.
# Obtain the expected checksum from a trusted source. "sha256sum FILE.deb" shows
# your local file's hash but is not independent proof of the package's origin.
# To use an approved public URL instead, comment out HARVESTER_DEB:
# HARVESTER_URL=https://approved-public-host.example/package.deb
# PEGASUS_BIN=/opt/PegasusHarvester/resources/app/node_modules/@nanometrics/pegasus-harvest-lib/build/Release/harvester

# TAILSCALE: a device enrollment AUTH KEY, not your Google/Tailscale login password.
# Generate one in the Tailscale admin console: Keys > Generate auth key.
# Copy the complete, case-sensitive generated key; keys start with "tskey-".
# The value below is deliberately not a working key.
TAILSCALE_AUTH_KEY=tskey-REPLACE_WITH_YOUR_GENERATED_DEVICE_AUTH_KEY
# Needed only for initial enrollment; omit if the computer is already enrolled.
# Prefer a one-use, appropriately tagged key. Tailnet policy must permit access.
# Do not substitute an API access token or OAuth client secret.
# Instructions: https://tailscale.com/docs/features/access-control/auth-keys

# VNC: choose a NEW, separate password yourself, not one issued by Google.
# This installer requires exactly EIGHT printable non-space ASCII characters.
# Example shape ONLY: "VncDemo1" (8 characters); NEVER deploy that public example.
ENABLE_VNC=yes
VNC_PASSWORD=REPLACE_WITH_YOUR_8_CHARACTER_VNC_PASSWORD
# With ENABLE_VNC=no, omit VNC_PASSWORD. VNC is reached through an SSH tunnel.
